```typescript
    const { stepId, messages, report, sessionId } = await req.json();

    // Convert ConvMsg to CoreMessage format

      role: m.role, // "user" | "assistant" | "system"
      content: m.content
    }));

    const result = streamText({
      model: anthropic.messages("claude-3-sonnet-20240229"),
      messages: coreMessages,
      system: STEP_SYSTEMS[stepId],
      tools: {
        generate_section: {
          description: "Generate a report section using the generation agent",
          parameters: z.object({
            sectionId: z.string(),
            context: z.string().optional()
          }),
          execute: async ({ sectionId, context }) => {
            // Pass full chat history to generation agent (fixes Bug 1)
            const fullContext = `
CHAT HISTORY (last 5 turns):
${messages.slice(-5).map((m: any) => `${m.role}: ${m.content}`).join('\n')}

EXTRA CONTEXT: ${context || ''}
CURRENT STEP: ${stepId}
            `.trim();

            const response = await fetch(`${process.env.BASE_URL}/api/session/${sessionId}/generate`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ sectionId, context: fullContext })
            });

            const data = await response.json();
            return { success: true, content: data.content };
          }
        },
        ask_user: {
          description: "Ask user to choose from options",
          parameters: z.object({
            question: z.string(),
            choices: z.array(z.string())
          }),
          execute: async ({ question, choices }) => ({ question, choices })
        },
        step_complete: {
          description: "Mark current step as complete and advance to next step",
          parameters: z.object({
            summary: z.string()
          }),
          execute: async ({ summary }) => ({ success: true, summary })
        }
      },
      maxTokens: 4000
    });

    return result.toAIStreamResponse();
  } catch (error) {
    console.error("/api/converse error:", error);
    return new Response(JSON.stringify({ error: "Stream failed" }), { status: 500 });
  }
}
```

## PHASE 2: Frontend — Replace useConversation Hook (1 hour)

File: artifacts/rapportai/src/hooks/use-conversation.ts

REPLACE entire file with:

```typescript
import { useChat } from "ai/react";
import { useReportStore } from "@/store/report-store"; // Your existing Zustand store

export function useConversation(stepId: string) {
  const { setGeneratedSections, advanceStep } = useReportStore();

  const {
    messages,
    input,
    handleInputChange,
    handleSubmit,
    isLoading,
    append, // For file uploads
    stop
  } = useChat({
    api: "/api/converse",
    body: { stepId }, // Sent with every request
    onFinish: (message) => {
      // Handle tool invocations automatically
      if (message.toolInvocations) {
        message.toolInvocations.forEach((tool: any) => {
          if (tool.toolName === "generate_section" && tool.result?.content) {
            setGeneratedSections({
              [tool.args.sectionId]: tool.result.content
            });
          }
          if (tool.toolName === "step_complete") {
            advanceStep(); // Moves to next step (e.g., step2 → step3)
          }
        });
      }
    },
    onError: (error) => {
      console.error("Chat stream error:", error);
      // Vercel SDK handles retries automatically
    }
  });

  return {
    messages,
    input,
    handleInputChange,
    handleSubmit,
    isLoading,
    append,
    stop
  };
}
```

```typescript
import { useConversation } from "@/hooks/use-conversation";
import { ChoiceCard } from "@/components/ChoiceCard";
import { useReportStore } from "@/store/report-store";

interface StepNPageProps {
  stepId: string; // e.g., "step2", "step3", ..., "step15"
  sectionId: string; // e.g., "page-de-garde", "remerciements", "resume", etc.
  title: string; // Page title
}

export default function StepNPage({ stepId, sectionId, title }: StepNPageProps) {
  const { messages, input, handleInputChange, handleSubmit, isLoading } = useConversation(stepId);
  const { generatedSections } = useReportStore();

  return (
    <div className="step-container">
      <h1>{title}</h1>
      
      <div className="chat-layout">
        {/* LEFT: Chat */}
        <div className="chat-panel">
          <div className="messages">
            {messages.map((msg) => (
              <div key={msg.id} className={`message ${msg.role}`}>
                <div className="message-content">
                  {msg.content}
                </div>
                
                {/* Render tool invocations (ask_user choices) */}
                {msg.toolInvocations?.map((tool: any) => {
                  if (tool.toolName === "ask_user" && tool.state === "result") {
                    return (
                      <ChoiceCard
                        key={tool.toolCallId}
                        question={tool.args.question}
                        choices={tool.args.choices}
                        onSelect={(choice) => {
                          append({ role: "user", content: choice });
                        }}
                      />
                    );
                  }
                })}
              </div>
            ))}
            
            {/* Loading indicator */}
            {isLoading && (
              <div className="loading">
                <div className="spinner"></div>
                <p>RapportAI travaille...</p>
              </div>
            )}
          </div>

          {/* Input form */}
          <form onSubmit={handleSubmit} className="input-form">
            <input
              value={input}
              onChange={handleInputChange}
              placeholder={`Ask about ${title.toLowerCase()}...`}
              disabled={isLoading}
            />
            <button type="submit" disabled={isLoading}>
              {isLoading ? "..." : "Send"}
            </button>
          </form>
        </div>

        {/* RIGHT: Preview (if section exists) */}
        <div className="preview-panel">
          {generatedSections[sectionId] ? (
            <div 
              className="section-preview"
              dangerouslySetInnerHTML={{ __html: generatedSections[sectionId] }}
            />
          ) : (
            <div className="no-content">
              <p>Generate content via chat to see preview here</p>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

```typescript
import { useState } from "react";

interface ChoiceCardProps {
  question: string;
  choices: string[];
  onSelect: (choice: string) => void;
}

export function ChoiceCard({ question, choices, onSelect }: ChoiceCardProps) {
  const [selected, setSelected] = useState<string | null>(null);

  return (
    <div className="choice-card">
      <p className="question">{question}</p>
      <div className="choices">
        {choices.map((choice) => (
          <button
            key={choice}
            onClick={() => {
              setSelected(choice);
              onSelect(choice);
            }}
            className={selected === choice ? "selected" : ""}
          >
            {choice}
          </button>
        ))}
      </div>
    </div>
  );
}
```

```typescript
export async function POST(req: Request, { params }: any) {
  const { sessionId } = params;
  const { sectionId, context } = await req.json();

  // context now includes chat history (from converse.ts)
  const systemPrompt = `
You are generating section: ${sectionId}
CONTEXT FROM CHAT HISTORY (user provided details in chat):
${context}

Generate the full section content in French (academic style for Moroccan students).
Use proper academic formatting, headings, and citations.
If the section is literature review, include references.
If the section is methodology, be detailed about tools and methods.
If the section is analysis, include data interpretation.
If the section is implementation, include code examples and screenshots placeholders.
  `.trim();

  // Spawn Claude Haiku for fast generation
  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-API-Key": process.env.ANTHROPIC_API_KEY,
      "anthropic-version": "2023-06-01"
    },
    body: JSON.stringify({
      model: "claude-3-haiku-20240307",
      messages: [{ role: "user", content: `Generate ${sectionId} section` }],
      system: systemPrompt,
      max_tokens: 4000
    })
  });

  const data = await response.json();
  const content = data.content.text;

  // Return SSE with section_content event
  return new Response(
    `event: section_content\ndata: ${JSON.stringify({ content })}\n\n`,
    { headers: { "Content-Type": "text/event-stream" } }
  );
}
```
